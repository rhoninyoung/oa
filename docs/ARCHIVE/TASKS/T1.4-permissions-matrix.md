## 实现要点

```typescript
export type PermissionAction =
  | 'read' | 'edit' | 'submit' | 'withdraw' | 'approve' | 'reject' | 'reschedule'
  | 'addRow' | 'deleteRow'

export interface PermissionContext {
  type: 'schedule' | 'master' | 'task'
  groupId: string
  ownGroup: boolean       // 当前用户是否属于该 group
  taskSource?: TaskSource  // 仅 master/task 场景需要
}

export function permissions(
  role: Role,
  ctx: PermissionContext,
  action: PermissionAction
): { allowed: true } | { allowed: false; code: string }
```
