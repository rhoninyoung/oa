  it('DT-TBL-01: normalizeRange works for all drag directions', () => {
    expect(normalizeRange({ r1: 5, c1: 3, r2: 2, c2: 1 }))
      .toEqual({ r1: 2, c1: 1, r2: 5, c2: 3 })
  })
})

describe('ClipboardManager', () => {
  it('DT-TBL-02: copy as TSV with quoted newlines', () => {
    expect(toTSV([{ v: 'a\nb' }])).toEqual('"a\nb"')
  })

  it('DT-TBL-03: paste small range into large → overflow returned', () => {
    expect(parsePaste('a\tb', { rows: 1, cols: 2 }, { rows: 3, cols: 3 }))
      .toMatchObject({ overflow: { rows: 0, cols: 1 } })
  })
})
