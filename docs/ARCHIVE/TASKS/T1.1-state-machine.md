## 实现要点

```typescript
// 错误码
export type TransitionErrorCode =
  | 'CONTENT_EMPTY'
  | 'ACTOR_NOT_OWNER'
  | 'ACTOR_NOT_PM'
  | 'REASON_INVALID'
  | 'INVALID_TRANSITION'

export interface TransitionResult {
  ok: true
} | {
  ok: false
  code: TransitionErrorCode
}

export function canTransition(
  from: ScheduleStatus,
  to: ScheduleStatus,
  actor: Role,
  ctx: { tasksNonEmpty: boolean; rejectReason?: string }
): TransitionResult
```
